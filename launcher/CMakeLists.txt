cmake_minimum_required(VERSION 3.20)
project(QBZDownloaderPortable LANGUAGES CXX)
add_executable(QBZDownloaderPortable WIN32 main.cpp)
target_compile_features(QBZDownloaderPortable PRIVATE cxx_std_17)
target_compile_definitions(QBZDownloaderPortable PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(QBZDownloaderPortable PRIVATE shell32 user32)
if(MSVC)
  target_compile_options(QBZDownloaderPortable PRIVATE /W4 /EHsc /utf-8)
  set_property(TARGET QBZDownloaderPortable PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
